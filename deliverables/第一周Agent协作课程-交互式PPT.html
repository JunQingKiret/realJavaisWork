public Item search(String keyword) {
    for (Item item : items) {
        if (item.name().contains(keyword)) {
            return item; // 第一条后立即结束
        }
    }
    return null;
}
public List<Item> search(String keyword) {
    String term = keyword == null
        ? "" : keyword.trim().toLowerCase();

    if (term.isBlank()) return List.of();

    return items.stream()
        .filter(item -> item.name()
            .toLowerCase().contains(term))
        .toList();
}
